cmake_minimum_required(VERSION 3.20)
project(sameloop_ui_lvgl VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(SAMELOOP_BUILD_UI "Build the LVGL/SDL2 desktop application" ON)
include(CTest)
add_library(sameloop_sim STATIC src/simulation.c)
target_include_directories(sameloop_sim PUBLIC src)
target_compile_options(sameloop_sim PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(sameloop_sim PUBLIC m)
if(BUILD_TESTING)
  add_executable(test_simulation tests/test_simulation.c)
  target_link_libraries(test_simulation PRIVATE sameloop_sim)
  add_test(NAME simulation COMMAND test_simulation)
endif()
if(SAMELOOP_BUILD_UI)
  find_path(SDL2_INCLUDE_DIR SDL.h PATH_SUFFIXES SDL2 REQUIRED)
  find_library(SDL2_LIBRARY NAMES SDL2 REQUIRED)
  add_library(sameloop_sdl2 INTERFACE)
  target_include_directories(sameloop_sdl2 INTERFACE "${SDL2_INCLUDE_DIR}")
  target_link_libraries(sameloop_sdl2 INTERFACE "${SDL2_LIBRARY}")
  include(FetchContent)
  set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "LVGL configuration" FORCE)
  set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
  set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
  set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "" FORCE)
  if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
  endif()
  # LVGL v9.2.2, immutable commit and checksum (not a moving branch).
  FetchContent_Declare(lvgl
    URL https://codeload.github.com/lvgl/lvgl/tar.gz/7f07a129e8d77f4984fff8e623fd5be18ff42e74
    URL_HASH SHA256=8a5e30223fd5b08cdeccb060f45b37a29868d6b1fdfc5bdbdf0dd89aaefb5a0e)
  FetchContent_MakeAvailable(lvgl)
  target_link_libraries(lvgl PUBLIC sameloop_sdl2)
  add_executable(sameloop-ui-lvgl src/main.c src/ui.c)
  target_compile_options(sameloop-ui-lvgl PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(sameloop-ui-lvgl PRIVATE sameloop_sim lvgl::lvgl sameloop_sdl2)
  if(BUILD_TESTING)
    add_test(NAME ui_smoke COMMAND sameloop-ui-lvgl --smoke-test)
    set_tests_properties(ui_smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy" TIMEOUT 20)
  endif()
endif()
